def test_factor_value():
    assert F == (1 + 30 / 26) ** -1.5 and abs(F - 0.31635738822500753) < 1e-17


def test_known_100_to_1_harmonic_and_geometric():
    h = I.interface_coefficients([100.0, 1.0], [0, 0.1, 0.2], "harmonic", factor=1.0)["interface_conductivity_m_s"][0]
    g = I.interface_coefficients([100.0, 1.0], [0, 0.1, 0.2], "geometric", factor=1.0)["interface_conductivity_m_s"][0]
    assert math.isclose(h, 200.0 / 101.0, rel_tol=1e-15) and math.isclose(g, 10.0, rel_tol=1e-14)
    assert h < g < 50.5                                          # harmonic < geometric < arithmetic


@pytest.mark.parametrize("method", I.METHODS)
def test_homogeneous_is_exact(method):
    k = 3.7e-5
    out = I.interface_coefficients([k, k, k], [0, 0.1, 0.2, 0.3], method)
    assert out["interface_conductivity_m_s"] == [F * k, F * k]   # exactly the local K, no rounding drift


def test_nominal_layers_match_two_kikj_and_sqrt():
    ks = [3.3e-5, 2.1e-5, 1.2e-5]
    k = [F * x for x in ks]
    h = I.interface_coefficients(ks, [0, 0.1, 0.2, 0.3], "harmonic")["interface_conductivity_m_s"]
    g = I.interface_coefficients(ks, [0, 0.1, 0.2, 0.3], "geometric")["interface_conductivity_m_s"]
    for i in range(2):
        assert math.isclose(h[i], 2 * k[i] * k[i + 1] / (k[i] + k[i + 1]), rel_tol=1e-14)
        assert math.isclose(g[i], math.sqrt(k[i] * k[i + 1]), rel_tol=1e-14)


def test_unequal_thickness_weights():
    w = I.interface_weights([0, 0.1, 0.4])                       # h = 0.1, 0.3
    assert w[0] == pytest.approx((0.25, 0.75), rel=1e-15) and math.isclose(sum(w[0]), 1.0)
    h = I.interface_coefficients([4.0, 1.0], [0, 0.1, 0.4], "harmonic", factor=1.0)["interface_conductivity_m_s"][0]
    g = I.interface_coefficients([4.0, 1.0], [0, 0.1, 0.4], "geometric", factor=1.0)["interface_conductivity_m_s"][0]
    assert math.isclose(h, 1.0 / (0.25 / 4.0 + 0.75 / 1.0), rel_tol=1e-15)
    assert math.isclose(g, 4.0 ** 0.25, rel_tol=1e-14)           # exp(0.25 ln4 + 0.75 ln1)
    # the thicker layer dominates, so the unequal-weight result differs from the equal-weight one
    assert not math.isclose(h, 2 * 4.0 / 5.0, rel_tol=1e-3)

def _scores(best: tuple, rmse: float = 0.004, ties: dict | None = None) -> dict:
    out = {}
    for s in S.training_scenarios():
        d, e = s["diffusion_coefficient"], s["evaporation_factor"]
        out[s["name"]] = {"overall_rmse": 0.01 + abs(d - best[0]) * 0.001 + abs(e - best[1]) * 0.001}
    out[S.scenario_name(*best)]["overall_rmse"] = rmse
    out.update(ties or {})
    return out


def test_selection_uses_open_only_and_tie_rule():
    sc = _scores((0.03, 0.5))
    sc["d0_E0_closed"] = {"overall_rmse": 1e-9}                       # closed controls never selected
    assert S.select_best(sc)["name"] == "d0.03_E0.5_open"
    sc["d0.1_E0.25_open"] = {"overall_rmse": 0.004}                   # exact tie: smaller E wins
    assert S.select_best(sc)["name"] == "d0.1_E0.25_open"
    sc["d0.3_E0.25_open"] = {"overall_rmse": 0.004}                   # same E: smaller d wins
    assert S.select_best(sc)["name"] == "d0.1_E0.25_open"
    sc["d0.01_E0.25_open"] = {"overall_rmse": 0.004}
    assert S.select_best(sc)["name"] == "d0.01_E0.25_open"
    sc2 = _scores((0.05, 1.0))
    sc2["d0.05_E1_open"] = {"overall_rmse": None}                     # undefined RMSE is ineligible
    assert S.select_best(sc2)["name"] != "d0.05_E1_open"
    with pytest.raises(S.ProtocolError):
        S.select_best({"d0_E0_open": {"overall_rmse": None}})


def test_heldout_union_dedup_and_order():
    bh, bg = S.select_best(_scores((0.03, 0.5))), S.select_best(_scores((0.1, 0.25)))
    names = [s["name"] for s in S.heldout_scenarios(bh, bg)]
    assert names == ["d0.03_E0.5_open", "d0.1_E0.25_open", "d0.05_E1.2_open", "d0_E0.5_open", "d0.03_E0_open", "d0_E0.25_open", "d0.1_E0_open", "d0_E0_open"]
    assert len(names) == len(set(names))
    same = [s["name"] for s in S.heldout_scenarios(bh, bh)]                          # identical selections de-duplicate
    assert same == ["d0.03_E0.5_open", "d0.05_E1.2_open", "d0_E0.5_open", "d0.03_E0_open", "d0_E0_open"]
    ref = S.select_best(_scores((0.05, 1.2)))                                         # selected pair equals the reference
    names = [s["name"] for s in S.heldout_scenarios(ref, ref)]
    assert names.count("d0.05_E1.2_open") == 1 and "d0_E1.2_open" in names and "d0.05_E0_open" in names
    zero = S.select_best(_scores((0.0, 0.5)))                                         # selected pair is the generic best: zero-exchange duplicate removed
    names = [s["name"] for s in S.heldout_scenarios(zero, zero)]
    assert names.count("d0_E0.5_open") == 1 and names.count("d0_E0_open") == 1

def _run(shift: float = 0.0, n: int = 3, L: int = 3) -> dict:
    base = np.linspace(0.02, 0.06, n * L).reshape(n, L)
    tr = {k: (base + shift).tolist() for k in ("water_m",)}
    tr.update({"percolation_m": np.zeros((n, L - 1)).tolist(), "diffusion_m": (np.full((n, L - 1), 1e-5) * (1 + shift)).tolist(), "evaporation_m": np.full(n, 1e-4).tolist(), "evaporation_unmet_m": np.zeros(n).tolist(),
               "drainage_m": np.zeros(n).tolist(), "surface_return_m": np.zeros(n).tolist(), "t_s": (86400.0 * np.arange(1, n + 1)).tolist()})
    return {"trajectory": tr, "layer_theta": (base + shift).tolist(), "initial_storage_m": 0.1, "final_storage_m": 0.09 + shift}


def _score(rmse: float) -> dict:
    pl = {v: {"n": 5, "bias": 0.001, "rmse": rmse, "mae": rmse, "mse": rmse**2} for v in ("501", "502", "503")}
    return {"per_layer": pl, "overall_rmse": rmse, "pooled_rmse_alternative": rmse, "skill_vs_initial_point_values_held": 0.2}


WIN = {"JORN4_A": ("train", "2025-04-05T21:00:00Z"), "JORN4_B": ("heldout", None), "MOAB2_C": ("heldout", None)}


def _meta(window: str, method: str) -> dict:
    """Result metadata of the driver (plan, sources, persistence) with this window's REAL pinned start/anchor."""
    w = S.load_template(window)["window"]
    role = S.WINDOWS[window]["role"]
    plan = {"name": f"{window} {role} site-derived hydraulics, {method} interface mixing (prescribed bounded sensitivity inputs)", "run_start_utc": w["run_start_utc"], "anchor_rule": w["anchor"], "model_days": 3,
            "initial_layer_water_m": [0.005, 0.006, 0.007], "initial_total_water_m": 0.018, "day_end_utc": ["a", "b", "c", "d"], "pet_mm_day": [4.0, 5.0, 6.0], "prepared_inputs": {"observed_end_theta_m3m3": {"501": [0.04, 0.03, 0.02]}}}
    src = {"maple_syrup": {"package_sha256": "37d8" + "0" * 60, "files": 180, "file": f"/tmp/{method}/maple_syrup/__init__.py"}, "maple": {"package_sha256": "fbc5" + "0" * 60, "files": 412, "file": f"/tmp/{method}/maple/__init__.py"}}
    return {"config_sha256": ("a" if method == "harmonic" else "b") * 64, "plan": plan, "sources_before": copy.deepcopy(src), "sources_after": copy.deepcopy(src),
            "persistence": {"initial_point_values_held": {"overall_rmse": 0.01}}}


def _results(shift_d: float = 1e-6, window: str = "JORN4_A"):
    names = ["d0_E0_open", "d0.05_E1.2_open", "d0.03_E0.5_open", "d0_E0.5_open"]
    rh = {**_meta(window, "harmonic"), "runs": {n: _run() for n in names}, "scores": {n: _score(0.01) for n in names}}
    rg = {**_meta(window, "geometric"), "runs": {n: _run(0.0 if n.startswith("d0_") else shift_d) for n in names}, "scores": {n: _score(0.01 if n.startswith("d0_") else 0.0102) for n in names}}
    return rh, rg

def _mutate(path, value):
    rh, rg = _results(window="JORN4_B")
    node = rg
    for k in path[:-1]:
        node = node[k]
    node[path[-1]] = value
    return rh, rg

def test_method_role_and_window_pairing_refused():
    rh, rg = _results(window="JORN4_B")
    with pytest.raises(S.ProtocolError):
        C.check_pair(rg, rh, window="JORN4_B")                                                       # methods swapped
    with pytest.raises(S.ProtocolError):
        C.check_pair(rh, rg, window="MOAB2_C")                                                       # requested window is not the results'
    with pytest.raises(S.ProtocolError):
        C.check_pair(rh, rg, window="JORN4_B", role="train")
    rm, rgm = _results(window="MOAB2_C")
    rgm["plan"]["name"] = rg["plan"]["name"]                                                         # a MOAB result labelled JORN4_B...
    rm["plan"]["name"] = rh["plan"]["name"]
    with pytest.raises(S.ProtocolError):
        C.check_pair(rm, rgm, window="JORN4_B")                                                      # ...fails the pinned start/anchor check
    rh2, rg2 = _results(window="JORN4_B")
    rg2["plan"]["name"] = "something else"
    with pytest.raises(S.ProtocolError):
        C.check_pair(rh2, rg2, window="JORN4_B")
    mix_h, mix_g = _results(window="JORN4_B")
    mix_g["plan"]["name"] = mix_g["plan"]["name"].replace("JORN4_B heldout", "MOAB2_C heldout")
    with pytest.raises(S.ProtocolError):
        C.check_pair(mix_h, mix_g, window="JORN4_B")


def test_heldout_window_wired_and_bound_to_config_hash():
    rh, rg = _results(window="JORN4_B")
    sel = _selection("JORN4_B", list(rh["runs"]))
    with pytest.raises(S.ProtocolError):
        C.compare_heldout(rh, rg, sel, window="MOAB2_C")
    with pytest.raises(S.ProtocolError):
        C.compare_heldout(rh, rg, sel)                                                               # strict mode demands the window
    rh["config_sha256"] = "c" * 64
    with pytest.raises(S.ProtocolError):
        C.compare_heldout(rh, rg, sel, window="JORN4_B")                                             # result not the frozen config
    rh, rg = _results(window="JORN4_B")
    sel["heldout_scenarios"] = sel["heldout_scenarios"][:-1]
    with pytest.raises(S.ProtocolError):
        C.compare_heldout(rh, rg, sel, window="JORN4_B")
    rh, rg = _results(window="JORN4_A")                                                              # a TRAIN pair is not accepted as held-out
    with pytest.raises(S.ProtocolError):
        C.compare_heldout(rh, rg, _selection("JORN4_A", list(rh["runs"])), window="JORN4_A")


def test_cli_compare_heldout_uses_window(tmp_path, capsys):
    rh, rg = _results(window="MOAB2_C")
    sel = _selection("MOAB2_C", list(rh["runs"]))
    for n, o in (("h", rh), ("g", rg), ("s", sel)):
        (tmp_path / f"{n}.json").write_text(json.dumps(o))
    base = ["compare-heldout", "--selection", str(tmp_path / "s.json"), "--harmonic", str(tmp_path / "h.json"), "--geometric", str(tmp_path / "g.json")]
    assert S.main(base + ["--window", "MOAB2_C"]) == 0
    capsys.readouterr()
    assert S.main(base + ["--window", "JORN4_B"]) == 2                                              # MOAB results are not accepted as JORN4_B
    assert "REFUSED" in capsys.readouterr().err


def _train_result(method: str, best=(0.03, 0.5), window: str = "JORN4_A"):
    names = [s["name"] for s in S.training_scenarios()]
    sc = {n: _score(v["overall_rmse"]) for n, v in _scores(best).items()}
    r = {**_meta(window, method), "runs": {n: _run() for n in names}, "scores": sc}
    return r


def test_select_best_rejects_malformed_rmse():
    for bad in (float("nan"), float("inf"), float("-inf"), -1e-9, True, "0.001", [0.1]):
        sc = _scores((0.03, 0.5))
        sc["d0.001_E0.1_open"] = {"overall_rmse": bad}
        with pytest.raises(S.ProtocolError):
            S.select_best(sc)
    sc = _scores((0.03, 0.5)); sc["d0_E0_closed"] = {"overall_rmse": float("nan")}                    # also refused for the non-selectable closed controls
    with pytest.raises(S.ProtocolError):
        S.select_best(sc)
    sc = _scores((0.03, 0.5)); sc["d0.001_E0.1_open"] = {"overall_rmse": None}                        # None keeps its meaning (undefined, skipped)
    assert S.select_best(sc)["name"] == "d0.03_E0.5_open"
    sc = _scores((0.03, 0.5)); sc["d0.001_E0.1_open"] = {"overall_rmse": 0.0}
    assert S.select_best(sc)["name"] == "d0.001_E0.1_open"


def test_finalize_verifies_actual_train_results(tmp_path):
    th, tg = _train_result("harmonic"), _train_result("geometric", (0.1, 0.25))
    th["config_sha256"], tg["config_sha256"] = "a" * 64, "b" * 64
    good = S.finalize(th, tg, tmp_path / "ok", {"harmonic": "a" * 64, "geometric": "b" * 64})
    assert good["common_primary_pair"] == "d0.03_E0.5_open"
    held = _train_result("harmonic"); held["plan"]["name"] = held["plan"]["name"].replace("JORN4_A train", "JORN4_B heldout")
    with pytest.raises(S.ProtocolError):
        S.finalize(held, tg, tmp_path / "e1")                                                          # a held-out result can never select
    with pytest.raises(S.ProtocolError):
        S.finalize(tg, th, tmp_path / "e2")                                                            # methods swapped
    short = _train_result("harmonic"); short["runs"].pop("d0_E0_closed")
    with pytest.raises(S.ProtocolError):
        S.finalize(short, tg, tmp_path / "e3")                                                         # not the frozen 56-scenario grid
    with pytest.raises(S.ProtocolError):
        S.finalize(th, tg, tmp_path / "e4", {"harmonic": "f" * 64, "geometric": "b" * 64})             # result not bound to the frozen config
    odd = _train_result("geometric"); odd["plan"]["pet_mm_day"] = [1.0, 1.0, 1.0]
    with pytest.raises(S.ProtocolError):
        S.finalize(th, odd, tmp_path / "e5")                                                           # different prepared inputs


def test_derived_local_conductivity_refused():
    with pytest.raises(ValueError):
        I.local_conductivities([5e-324])                                                               # positive denormal: f * 5e-324 underflows to 0
    with pytest.raises(ValueError):
        I.interface_coefficients([5e-324, 1e-5], [0, 0.1, 0.2], "harmonic")                            # no ZeroDivisionError
    with pytest.raises(ValueError):
        I.interface_coefficients([1e-5, 5e-324], [0, 0.1, 0.2], "geometric")
    with pytest.raises(ValueError):
        I.local_conductivities([1e308], factor=1e10)                                                   # overflow to inf
    assert I.local_conductivities([1e-300])[0] > 0.0
